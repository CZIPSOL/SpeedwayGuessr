rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- Funkcje pomocnicze (More/Less) ---
    function mlOwnerOrGuest(playerId) {
      return (request.auth != null && playerId == request.auth.uid) || playerId.matches('guest_.*');
    }
    // wpis okresowy: suma punktów z gier Daily (np. 23/28), liczba gier, mapa dni
    function mlPeriodOk() {
      let d = request.resource.data;
      return d.score >= 0
        && d.score <= d.total
        && d.total >= 1
        && d.total <= 20000
        && d.games is number && d.games >= 1 && d.games <= 1000
        && d.days is map;
    }

    // 1. Profil Gracza
    match /users/{userId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // 2. Ranking Daily
    match /rankings/{timeId}/scores/{playerId} {
      allow read: if true;
      allow write: if
        ((request.auth != null && playerId == request.auth.uid) || playerId.matches('guest_.*'))
        && request.resource.data.guesses >= 1
        && request.resource.data.guesses <= 10
        && (request.resource.data.won == 0 || request.resource.data.won == 1);
    }

    // 3. Rankingi Klasyczne (Daily, Tygodniowe, Miesięczne, All-time)
    match /{rankingType}/{timeId}/scores/{playerId} {
      allow read: if true;
      allow write: if rankingType.matches('leaderboard_.*') &&
        ((request.auth != null && playerId == request.auth.uid) || playerId.matches('guest_.*'))
        && (
          resource == null ||
          (
            request.resource.data.wins <= resource.data.get('wins', 0) + 1 &&
            request.resource.data.guesses <= resource.data.get('guesses', 0) + 10
          )
        );
    }

    // 4. More/Less - ranking dzienny
    // POPRAWKA: wcześniej score/total <= 12, a gra ma 14 rund, więc KAŻDY zapis był odrzucany.
    // Usunięty też limit day <= 31 (od dnia 32, czyli 24.11, zapisy by przestały działać).
    match /rankings_moreless/{dayId}/scores/{playerId} {
      allow read: if true;
      allow write: if mlOwnerOrGuest(playerId)
        && request.resource.data.score >= 0
        && request.resource.data.score <= request.resource.data.total
        && request.resource.data.total >= 1
        && request.resource.data.total <= 100
        && request.resource.data.day >= 1
        && request.resource.data.mode == 'daily';
    }

    // 4b. More/Less - rankingi tygodniowy / miesięczny / ogólny (NOWE)
    // Ogólna reguła nr 3 pasuje do tych kolekcji (leaderboard_...), ale blokuje aktualizację,
    // bo wymaga pól wins/guesses, których tu nie ma - stąd osobne, wyraźne reguły.
    match /leaderboard_moreless_weekly/{periodId}/scores/{playerId} {
      allow read: if true;
      allow write: if mlOwnerOrGuest(playerId) && mlPeriodOk();
    }
    match /leaderboard_moreless_monthly/{periodId}/scores/{playerId} {
      allow read: if true;
      allow write: if mlOwnerOrGuest(playerId) && mlPeriodOk();
    }
    match /leaderboard_moreless_alltime/{periodId}/scores/{playerId} {
      allow read: if true;
      allow write: if mlOwnerOrGuest(playerId) && mlPeriodOk();
    }
    // Stara kolekcja leaderboard_moreless: gra v4.19+ już z niej nie korzysta.
    // Reguła zostaje (z poprawionymi limitami), żeby starsza wersja gry w przeglądarkach graczy nie dostawała błędów.
    match /leaderboard_moreless/{playerId} {
      allow read: if true;
      allow write: if mlOwnerOrGuest(playerId)
        && request.resource.data.score >= 0
        && request.resource.data.score <= request.resource.data.total
        && request.resource.data.total >= 1
        && request.resource.data.total <= 100
        && request.resource.data.day >= 1
        && request.resource.data.mode == 'daily';
    }

    // 5. Kolejka Clash
    match /clash_queue/{docId} {
      allow read, write: if true;
    }

    // 6. Pokoje Clash
    match /clash_rooms/{roomId} {
      allow read, write: if true;
    }

    // 7. Ranking Ligowy ELO
    match /leaderboard_clash_beta/{userId} {
      allow read: if true;
      allow write: if
        ((request.auth != null && userId == request.auth.uid) || userId.matches('guest_.*'))
        && request.resource.data.elo <= 5000
        && (
          resource == null ||
          (
            request.resource.data.elo <= resource.data.get('elo', 1000) + 200 &&
            request.resource.data.elo >= resource.data.get('elo', 1000) - 200
          )
        );
    }

    // 8. Ranking TIME ATTACK
    match /leaderboard_timeattack/{playerId} {
      allow read: if true;
      allow write: if
        ((request.auth != null && playerId == request.auth.uid) || playerId.matches('guest_.*'))
        && request.resource.data.score >= 0
        && request.resource.data.score <= 1000
        && (
          resource == null ||
          request.resource.data.score >= resource.data.get('score', 0)
        );
    }

    // 9. Zgłaszanie brakujących zawodników
    match /player_suggestions/{docId} {
      allow read: if false;
      allow create: if request.resource.data.playerName.size() < 100
                    && request.resource.data.notes.size() < 500;
      allow update, delete: if false;
    }

    // 10. Zgłaszanie błędów
    match /bug_reports/{docId} {
      allow read: if false;
      allow create: if request.resource.data.description.size() < 2000;
      allow update, delete: if false;
    }
  }
}
